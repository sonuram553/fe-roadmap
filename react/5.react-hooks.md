## 1. useEffect

**Purpose**: Manages side effects in functional components (data fetching, subscriptions, DOM manipulation, cleanup)

**Basic syntax**:
```javascript
useEffect(() => {
  // Side effect code
  return () => {
    // Cleanup code (optional)
  };
}, [dependencies]); // Dependencies array
```

**Examples**:

```javascript
import { useEffect, useState } from 'react';

function UserProfile({ userId }) {
  const [user, setUser] = useState(null);

  // Effect runs on mount and when userId changes
  useEffect(() => {
    const fetchUser = async () => {
      const response = await fetch(`/api/users/${userId}`);
      const userData = await response.json();
      setUser(userData);
    };

    fetchUser();
  }, [userId]); // Dependency array

  // Effect with cleanup (for subscriptions)
  useEffect(() => {
    const timer = setInterval(() => {
      console.log('Timer tick');
    }, 1000);

    // Cleanup function
    return () => clearInterval(timer);
  }, []); // Empty deps = runs once on mount

  return <div>{user?.name}</div>;
}
```

**Dependency patterns**:
- `[]` - Runs once on mount
- `[value]` - Runs when `value` changes
- No array - Runs on every render (usually avoided)

## 2. useMemo

**Purpose**: Memoizes expensive calculations to prevent unnecessary re-computations

**Basic syntax**:
```javascript
const memoizedValue = useMemo(() => {
  return expensiveCalculation(a, b);
}, [a, b]);
```

**Examples**:

```javascript
import { useMemo, useState } from 'react';

function ExpensiveComponent({ items, filter }) {
  const [count, setCount] = useState(0);

  // Expensive filtering operation
  const filteredItems = useMemo(() => {
    console.log('Filtering items...'); // Only logs when items/filter change
    return items.filter(item => 
      item.name.toLowerCase().includes(filter.toLowerCase())
    );
  }, [items, filter]);

  // Complex calculation
  const expensiveValue = useMemo(() => {
    return items.reduce((sum, item) => sum + item.price, 0);
  }, [items]);

  return (
    <div>
      <p>Count: {count}</p>
      <button onClick={() => setCount(count + 1)}>
        Increment (won't recalculate filtered items)
      </button>
      <p>Total: ${expensiveValue}</p>
      <ul>
        {filteredItems.map(item => (
          <li key={item.id}>{item.name}</li>
        ))}
      </ul>
    </div>
  );
}
```

## 3. useCallback

**Purpose**: Memoizes function definitions to prevent child component re-renders

**Basic syntax**:
```javascript
const memoizedCallback = useCallback(() => {
  // Function logic
}, [dependencies]);
```

**Examples**:

```javascript
import { useCallback, useState, memo } from 'react';

// Child component wrapped in memo
const Button = memo(({ onClick, children }) => {
  console.log('Button rendered');
  return <button onClick={onClick}>{children}</button>;
});

function Parent() {
  const [count, setCount] = useState(0);
  const [name, setName] = useState('');

  // Without useCallback, this function is recreated on every render
  // causing Button to re-render even when count doesn't change
  const handleIncrement = useCallback(() => {
    setCount(prev => prev + 1);
  }, []); // No dependencies needed for setter function

  // Callback with dependencies
  const handleNamedIncrement = useCallback(() => {
    console.log(`Incrementing for ${name}`);
    setCount(prev => prev + 1);
  }, [name]); // Recreates when name changes

  return (
    <div>
      <p>Count: {count}</p>
      <input 
        value={name} 
        onChange={(e) => setName(e.target.value)}
        placeholder="Enter name"
      />
      
      {/* These buttons won't re-render when name changes */}
      <Button onClick={handleIncrement}>
        Increment
      </Button>
      
      {/* This button re-renders only when name changes */}
      <Button onClick={handleNamedIncrement}>
        Named Increment
      </Button>
    </div>
  );
}
```

## 4. useRef

**Purpose**: 
1. Access DOM elements directly
2. Store mutable values that don't trigger re-renders
3. Keep references to values across renders

**Basic syntax**:
```javascript
const ref = useRef(initialValue);
// Access with ref.current
```

**Examples**:

```javascript
import { useRef, useEffect, useState } from 'react';

function MultiPurposeRef() {
  const [count, setCount] = useState(0);
  
  // 1. DOM reference
  const inputRef = useRef(null);
  const divRef = useRef(null);

  // 2. Mutable value that doesn't cause re-renders
  const renderCountRef = useRef(0);
  
  // 3. Previous value storage
  const prevCountRef = useRef();

  useEffect(() => {
    // Track render count
    renderCountRef.current += 1;
    
    // Store previous count
    prevCountRef.current = count;
  });

  const focusInput = () => {
    inputRef.current?.focus();
  };

  const scrollToDiv = () => {
    divRef.current?.scrollIntoView({ behavior: 'smooth' });
  };

  return (
    <div>
      <p>Current count: {count}</p>
      <p>Previous count: {prevCountRef.current}</p>
      <p>Render count: {renderCountRef.current}</p>
      
      <input ref={inputRef} placeholder="Focus me!" />
      <button onClick={focusInput}>Focus Input</button>
      
      <button onClick={() => setCount(count + 1)}>
        Increment
      </button>
      
      <div style={{ height: '200px' }}>Spacer</div>
      <div ref={divRef}>Scroll target</div>
      <button onClick={scrollToDiv}>Scroll to Div</button>
    </div>
  );
}

// Timer example with useRef
function Timer() {
  const [time, setTime] = useState(0);
  const intervalRef = useRef(null);

  const startTimer = () => {
    if (intervalRef.current) return; // Already running
    
    intervalRef.current = setInterval(() => {
      setTime(prev => prev + 1);
    }, 1000);
  };

  const stopTimer = () => {
    if (intervalRef.current) {
      clearInterval(intervalRef.current);
      intervalRef.current = null;
    }
  };

  const resetTimer = () => {
    stopTimer();
    setTime(0);
  };

  // Cleanup on unmount
  useEffect(() => {
    return () => stopTimer();
  }, []);

  return (
    <div>
      <p>Time: {time}s</p>
      <button onClick={startTimer}>Start</button>
      <button onClick={stopTimer}>Stop</button>
      <button onClick={resetTimer}>Reset</button>
    </div>
  );
}
```

## When to Use Each Hook

**useEffect**:
- Data fetching
- Setting up subscriptions
- DOM manipulation
- Cleanup (removing event listeners, clearing timers)

**useMemo**:
- Expensive calculations
- Complex object/array transformations
- When child components depend on computed values

**useCallback**:
- Passing callbacks to optimized child components
- Dependencies for other hooks
- Event handlers that shouldn't change often

**useRef**:
- Accessing DOM elements
- Storing mutable values without re-renders
- Keeping references to timers, intervals
- Storing previous values

## Performance Tips

1. **Don't overuse memoization** - Only optimize when you have performance issues
2. **useCallback and useMemo have overhead** - Use them judiciously
3. **Dependencies matter** - Always include all dependencies in arrays
4. **useRef doesn't trigger re-renders** - Perfect for values you need to mutate

These hooks form the foundation of modern React development and help you build efficient, maintainable components!
